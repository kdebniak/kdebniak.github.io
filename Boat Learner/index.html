let quizData = []; let currentIdx = 0; let score = 0; let answered = false; // NEW: Fetch questions from the external JSON file async function loadQuizData() { try { const response = await fetch('questions.json'); quizData = await response.json(); init(); // Start the quiz once data is loaded } catch (error) { console.error("Error loading questions:", error); document.getElementById('question-text').innerText = "Failed to load questions. Check console."; } } function init() { if (quizData.length === 0) return; answered = false; const data = quizData[currentIdx]; document.getElementById('question-text').innerText = data.q; document.getElementById('q-count').innerText = `Question ${currentIdx + 1} of ${quizData.length}`; document.getElementById('progress-fill').style.width = `${(currentIdx / quizData.length) * 100}%`; const optionsDiv = document.getElementById('options'); optionsDiv.innerHTML = ''; data.a.forEach((opt, i) => { const b = document.createElement('button'); b.className = 'option-btn'; b.innerText = opt; b.onclick = () => { if(!answered) handleAns(i, b); }; optionsDiv.appendChild(b); }); document.getElementById('next-btn').style.display = 'none'; } function handleAns(i, b) { answered = true; const correct = quizData[currentIdx].c; if (i === correct) { b.classList.add('correct'); score++; } else { b.classList.add('wrong'); document.querySelectorAll('.option-btn')[correct].classList.add('correct'); } document.getElementById('next-btn').style.display = 'inline-block'; } function nextQuestion() { currentIdx++; if (currentIdx < quizData.length) init(); else finish(); } function finish() { document.getElementById('question-content').classList.add('hidden'); document.getElementById('quiz-header').classList.add('hidden'); document.getElementById('result-screen').style.display = 'block'; document.getElementById('final-score').innerText = `${score} / ${quizData.length}`; let msg = score > 12 ? "Master Mariner! You're ready to set sail." : "Good effort! Review the rules and try again."; document.getElementById('feedback').innerText = msg; } // Start the process loadQuizData();
